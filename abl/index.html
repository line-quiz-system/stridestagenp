/** * StrideStage フロントエンド (GitHub Pages / index.html 内スクリプト) 改修用コード * * 【改修ポイント】 * 1. `liff.getAccessToken()` を取得し、GASへの全リクエスト（クイズ取得・解約ポータル取得）に付与します。 * これにより、第三者がURLパラメータの `uid` を書き換えて他人のアカウントやStripe解約画面に * なりすます攻撃（IDOR脆弱性）を完全に遮断します。 * 2. 解約・プラン確認ポータルURLの取得処理を `openCustomerPortal()` に1本化しました。 */ // ========================================== // 1. クイズ＆論文データ取得処理（LIFF初期化後） // ========================================== async function fetchQuizAndArticles(gasWebAppUrl, liffId) { await liff.init({ liffId: liffId }); if (!liff.isLoggedIn()) { liff.login(); return; } // ★重要: 本人証明用のLIFFアクセストークンを取得 const accessToken = liff.getAccessToken(); const profile = await liff.getProfile(); const uid = profile.userId; // URLパラメータ ?d=XX の引き継ぎ const urlParams = new URLSearchParams(window.location.search); const dayParam = urlParams.get("d"); // GASへ accessToken と uid の両方を送信 const requestUrl = new URL(gasWebAppUrl); requestUrl.searchParams.set("accessToken", accessToken); requestUrl.searchParams.set("uid", uid); if (dayParam !== null && dayParam !== "") { requestUrl.searchParams.set("d", dayParam); } const response = await fetch(requestUrl.toString(), { method: "GET", mode: "cors" }); const data = await response.json(); if (!data.isAuth || data.error) { // エラー表示処理（既存のエラー描画関数を呼び出し） console.warn("Access denied or expired:", data.error); return data; } // data.quizData (3問) と data.articleData (論文3件) を画面に描画 return data; } // ========================================== // 2. 解約・プラン確認ポータル遷移処理（統合版） // ========================================== async function openCustomerPortal(gasWebAppUrl) { try { if (!liff.isLoggedIn()) { alert("LINEアプリ内からアクセスしてください。"); return; } const accessToken = liff.getAccessToken(); const profile = await liff.getProfile(); // GET (?action=getPortalUrl) で統一呼び出し（CORSエラーが起きにくく最速） const portalRequestUrl = new URL(gasWebAppUrl); portalRequestUrl.searchParams.set("action", "getPortalUrl"); portalRequestUrl.searchParams.set("accessToken", accessToken); portalRequestUrl.searchParams.set("userId", profile.userId); const response = await fetch(portalRequestUrl.toString(), { method: "GET", mode: "cors" }); const result = await response.json(); if ((result.success || result.status === "success") && result.url) { // Stripeカスタマーポータルへリダイレクト window.location.href = result.url; } else { alert(result.error || result.message || "解約・プラン確認ページの取得に失敗しました。"); } } catch (err) { console.error("Portal URL Error:", err); alert("通信エラーが発生しました。時間を置いて再度お試しください。"); } }
